# 一括検証 2 回目（side20）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section clean349a
do kill @e[type=!player,family=!skin_studio]

section clean349c

section mobs-zombie_pigman

do effect @e[type=zombie_pigman] slowness 60 255 true
do effect @e[type=zombie_pigman] fire_resistance 60 1 true
do effect @e[type=zombie_pigman] water_breathing 60 1 true
do tp @a -2.5 -60 5.5 -25 3
wait 3500
shot mobs-zombie_pigman-mirror                       # zombie_pigman: 鏡の中で左右反転・同じ模様・同じ大きさ・同じ動き

do execute as @a at @s run studio:remove

do tp @a -2.5 -60 5.5 -25 3
do camera @a set minecraft:free pos -2.5 -58.38 18.5 rot 3 -155
wait 2500
shot mobs-zombie_pigman-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して mobs-zombie_pigman-mirror と比べる）
