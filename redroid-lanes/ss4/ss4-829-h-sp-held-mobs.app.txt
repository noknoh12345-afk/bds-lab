# 一括検証 4 回目（h-sp-held-mobs）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section reset
do time set 6000
do weather clear
do camera @a clear
do setblock 7 -54 -3 stone
do fill -7 -60 -3 7 -54 11 air
section held
do summon zombie -0.5 -60 8.5 150 0
do summon skeleton 1.5 -60 7.5 150 0
do replaceitem entity @e[type=zombie] slot.weapon.mainhand 0 iron_sword
do replaceitem entity @e[type=zombie] slot.armor.head 0 golden_helmet
do replaceitem entity @e[type=skeleton] slot.armor.chest 0 diamond_chestplate
do effect @e[type=!player,family=!skin_studio] slowness 60 255 true
do effect @e[type=!player,family=!skin_studio] fire_resistance 60 1 true
do tp @a 2.5 -60 5.5 20 3
wait 4000
shot held-mirror                          # 持ち物・防具が映る
section held-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air
do tp @a 2.5 -60 5.5 20 3
do camera @a set minecraft:free pos 2.5 -58.38 18.5 rot 3 160
wait 2500
shot held-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して held-mirror と比べる）
do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
section done
