do effect @e[type=copper_golem] slowness 60 255 true
do effect @e[type=copper_golem] fire_resistance 60 1 true
do effect @e[type=copper_golem] water_breathing 60 1 true
do tp @a 2.5 -60 5.5 20 3
wait 3500
shot moban-copper_golem-mirror                       # copper_golem: 鏡の中で左右反転・同じ模様・同じ大きさ・同じ動き
section moban-copper_golem-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air
do tp @a 2.5 -60 5.5 20 3
do camera @a set minecraft:free pos 2.5 -58.38 18.5 rot 3 160
wait 2500
shot moban-copper_golem-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して moban-copper_golem-mirror と比べる）
do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
