do effect @e[type=happy_ghast] fire_resistance 60 1 true
do effect @e[type=happy_ghast] water_breathing 60 1 true
do tp @a 4.5 -60 6.5 35 8
wait 3500
record start
wait 6000
record stop                               # happy_ghast: 自由に動く。本物と鏡の像が同じ動き

wait 500
shot mobcw-happy_ghast-mirror                       # happy_ghast: 鏡の中で左右反転・同じ模様・同じ大きさ・同じ動き
section mobcw-happy_ghast-truth

do execute as @a at @s run studio:remove

wait 2500

do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
