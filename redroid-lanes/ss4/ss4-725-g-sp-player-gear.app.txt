# 一括検証 4 回目（g-sp-player-gear）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section reset
do time set 18000
do weather clear
do camera @a clear
do fill -7 -60 -3 7 -54 11 air
section gear
do replaceitem entity @a slot.weapon.mainhand 0 diamond_sword
do replaceitem entity @a slot.weapon.offhand 0 shield
do replaceitem entity @a slot.armor.head 0 iron_helmet
do replaceitem entity @a slot.armor.chest 0 elytra
do tp @a 0.5 -60 8 0 0
wait 3000
shot gear-front
do camera @a set minecraft:free pos 3 -58.3 6 facing 0.5 -59 12
wait 2000
shot gear-side                            # 自分の剣・盾・兜・エリトラが鏡に映る
do camera @a clear
section done
