# 一括検証 4 回目（i-sp-mirror-in-mirror）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section reset
do time set 22000
do weather clear
do camera @a clear
do setblock 7 -54 -3 stone
do fill -7 -60 -3 7 -54 11 air
section mim
do fill -4 -60 0 4 -56 0 stone
do fill -1 -60 0 1 -58 0 air
do studio:mirror 0 -60 0 south
do tp @a 0.5 -60 6 0 0
wait 5000
shot mim-north
do tp @a 0.5 -60 6 180 0
wait 3000
shot mim-south                            # 向かい合う鏡: 鏡の中の鏡が、重さが許す深さまで続く
section done
