do effect @e[type=pillager] fire_resistance 60 1 true
do effect @e[type=pillager] water_breathing 60 1 true
do tp @a 2.5 -60 5.5 20 3
wait 3500
record start
wait 6000
record stop                               # pillager: 自由に動く。本物と鏡の像が同じ動き
shot mobgw-pillager-mirror                       # pillager: 鏡の中で左右反転・同じ模様・同じ大きさ・同じ動き
section mobgw-pillager-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air
do tp @a 2.5 -60 5.5 20 3
do camera @a set minecraft:free pos 2.5 -58.38 18.5 rot 3 160
wait 2500
shot mobgw-pillager-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して mobgw-pillager-mirror と比べる）
do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
