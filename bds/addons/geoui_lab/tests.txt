## player にプロパティを32個まで宣言でき、すべて既定値で読める
@A join
js const e = p('A'); const ids = ['bar','big','f','on','mode','size', ...Array.from({length: 26}, (_, i) => 'p' + String(i + 6).padStart(2, '0'))]; JSON.stringify([ids.length, ids.filter((k) => e.getProperty('glab:' + k) === undefined).length])
= [32,0]
js const e = p('A'); JSON.stringify([e.getProperty('glab:bar'), e.getProperty('glab:big'), e.getProperty('glab:on'), e.getProperty('glab:mode'), e.getProperty('glab:p31')])
= [50,0,true,"a",0]

## setProperty の値は同じ tick の getProperty には出ず、次の tick から読める
js const e = p('A'); e.setProperty('glab:bar', 7); e.getProperty('glab:bar')
= 50
wait 100
js p('A').getProperty('glab:bar')
= 7

## int の範囲外は丸めずに例外（ArgumentOutOfBoundsError）
js try { p('A').setProperty('glab:bar', 101); return 'ok' } catch (x) { return String(x) }
~ ArgumentOutOfBoundsError: .*Value: 101, Argument bounds: \[0, 100\]
js try { p('A').setProperty('glab:bar', -1); return 'ok' } catch (x) { return String(x) }
~ ArgumentOutOfBoundsError

## int に小数を渡すと 0 の方へ切り捨て

js p('A').setProperty('glab:bar', -0.4); 0

## プロパティはプレイヤーごと（A を変えても B は既定値のまま）
@B join
@A cmd /glab:bar 42
~ @A GLAB bar 42
wait 100
js JSON.stringify([p('A').getProperty('glab:bar'), p('B').getProperty('glab:bar')])
= [42,50]

## コマンドで書いた値を1tick後に読める（権限 Any のコマンド）
@B cmd /glab:bar 9
~ @B GLAB bar 9
@B cmd /glab:get
~ @B GLAB bar=9 on=true
@B cmd /glab:bar 101
~ @B GLAB bar must be 0\.\.100 \(got 101\)
@B cmd /glab:show false
~ @B GLAB show false
@B cmd /glab:get
~ @B GLAB bar=9 on=false

## 大きさ（/glab:size）は 10〜1000 % だけ受け付ける

~ @A GLAB size 400
@A cmd /glab:size 5
~ @A GLAB size must be 10\.\.1000 \(got 5\)
wait 100
js p('A').getProperty('glab:size')
= 400

## スコアボードの値が2tick以内にプロパティへ写る（0〜100 に収める）
