# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音そのものは `audio export` が Opus にして実行の成果物の
#   public/spectral/rec/ に公開する（notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。1 録音 = 1 区切り（失敗はその録音だけ）
# キュー 97（tools/exp/queue/q97_speech_duck_renew.json、11-18）: q95 と同じ手順（ダッキング 0 / 6 / 12 dB と曲なし、3 文）に、読み上げの後にライブの音を作り直す renew（1.5.12、既定で入）。q92・q95・q96 で電話が落としたライブの音が、読み上げの後に戻るか。読み上げの後 4 秒長く録る

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song rock_big suffix=_e pack=tools/exp/packs/run10
audio song speech_hello,speech_weather,speech_piano pack=tools/exp/packs/notes

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear
do execute as @a run tp @s 0 -60 0 0 0

section rn_none_hello
do scriptevent spectral:room 0
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600 true true
do scriptevent spectral:duck 0
audio start rnnone_hello
do execute as @a run spectral:play speech_hello
wait 11600
do execute as @a run spectral:stop
audio stop
audio export rnnone_hello 160

section rn_0_hello
do scriptevent spectral:room 0
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600 true true
do scriptevent spectral:duck 0
audio start rn0_hello
do execute as @a run spectral:live_demo rock_big_e near @s 24 "near:4,-58.5,0,overworld"
wait 1000
do execute as @a run spectral:play speech_hello
wait 11600
do execute as @a run spectral:stop
do execute as @a run spectral:live demo_rock_big_e stop
audio stop
audio export rn0_hello 160
