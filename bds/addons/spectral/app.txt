# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 19 回目（tools/exp/make_app.py tools/exp/app19.json）: 作り直した音が鳴り出すまで声が抜ける（18 回目: 0.3–0.4 秒）のを直す試し。
#   どれも定義の音量 0.01（新しい既定）。ya_ / yb_ = 止まった話し手 0.01 / 0.1。歩く話し手（17・18 回目と同じ道）: yc_ / yj_ = 1.2.0 の引き継ぎ
#   （hold、2 tick）、yd_ / yi_ = その後 2 tick ごとに状態を 3 回送り直す、ye_ / yh_ = 古い音も声に付いて 6 tick 後に引き継ぐ（steer）、
#   yf_ / yg_ = その両方（spectral:handover <hold|steer> 1.5 <tick> <送り直す回数>）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section ya
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true 0.01
do execute as @a run spectral:handover hold 1.5 2 0
do execute as @a run summon armor_stand talker 0 -60 3
audio start ya_1
audio mark ya_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d follow @s 24 "follow:talker"
wait 23000
audio stop
audio analyze-all ya_speech_ja_soseki=song:speech_ja_soseki
audio export ya_speech_ja_soseki 160
