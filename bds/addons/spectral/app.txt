# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音は app/runs/<実行>/audio/*.wav（成果物には入らない）。
#   数はジョブのログの `[spectral rig] <録音> <何>: {...}` の行（audio analyze）。録音そのものは `audio export` が Opus にして
#   tools/rig/rec_public.pem で暗号化し `[spectral rec]` の行でログへ（開けるのはその秘密鍵の持ち主だけ: tools/rig/rec_decode.mjs）。
# いまは原因を見る短い版（run 41055135707: デモが 1 つの音の高さに集まり、最初の試験曲は無音だった）: 較正・デモ・試験曲 2 曲
#   （再生はコンソールの execute as で: チャットの打ち込みを外す）・軽量モード 1 曲・天井 2 曲を、どれも書き出す。

section title
launch
until title 600000
shot title

section prep
audio song piano_gymnopedie
audio song rock_big
audio ceiling spectral_ceil piano_gymnopedie,rock_big

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section calib
audio start calib
do spectral:calib all
until server spectral calib\] end throughput 300000
wait 2000
audio stop
audio analyze calib all
audio export calib 96

section demo
audio start demo
do execute as @a run spectral:play demo
wait 27000
audio stop
audio analyze demo song:demo
audio export demo 160

section songs
audio start g_piano_gymnopedie
do execute as @a run spectral:play piano_gymnopedie
wait 23000
audio stop
audio analyze g_piano_gymnopedie song:piano_gymnopedie
audio analyze g_piano_gymnopedie emu:piano_gymnopedie
audio export g_piano_gymnopedie 160
audio start g_rock_big
do execute as @a run spectral:play rock_big
wait 23000
audio stop
audio analyze g_rock_big song:rock_big
audio analyze g_rock_big emu:rock_big
audio export g_rock_big 160

section light
do execute as @a run spectral:mode light
audio start l_rock_big
do execute as @a run spectral:play rock_big
wait 23000
audio stop
audio analyze l_rock_big song:rock_big
audio export l_rock_big 160
do execute as @a run spectral:mode quality

section ceiling
addon spectral_ceil
join
until joined 300000
wait 5000
audio start c_piano_gymnopedie
do scriptevent spectral:ceiling piano_gymnopedie
wait 22000
audio stop
audio analyze c_piano_gymnopedie song:piano_gymnopedie
audio export c_piano_gymnopedie 160
audio start c_rock_big
do scriptevent spectral:ceiling rock_big
wait 22000
audio stop
audio analyze c_rock_big song:rock_big
audio export c_rock_big 160
