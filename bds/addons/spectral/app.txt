# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音そのものは `audio export` が Opus にして実行の成果物の
#   public/spectral/rec/ に公開する（notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。1 録音 = 1 区切り（失敗はその録音だけ）
# キュー 47（tools/exp/queue/q47_speech_over_music.json、tools/exp/queue40.py）: 曲が鳴っている中で読み上げ。4 ブロック先で曲（正弦波の道）を流しながら読み上げ: 32 音の取り合いで素が落ちるか、聞き取りはどれだけ下がるか

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song rock_big suffix=_e pack=tools/exp/packs/run10
audio song speech_hello,speech_weather,speech_cave,speech_piano pack=tools/exp/packs/notes

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear
do execute as @a run tp @s 0 -60 0 0 0

section ovr_none_hello
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
audio start ov0_speech_hello
do execute as @a run spectral:play speech_hello
wait 7600
do execute as @a run spectral:stop
audio stop
audio export ov0_speech_hello 160

section ovr_music_hello
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
audio start ovm_speech_hello
do execute as @a run spectral:live_demo rock_big_e near @s 24 "near:4,-58.5,0,overworld"
wait 1000
do execute as @a run spectral:play speech_hello
wait 7600
do execute as @a run spectral:stop
do execute as @a run spectral:live demo_rock_big_e stop
audio stop
audio export ovm_speech_hello 160
