# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 26 回目（tools/exp/make_app.py tools/exp/app26.json）: 5 回目のセッションの最後の既定をそのまま（設定の命令なし）で通しに: fa_ = 曲をヘッドホンで
#   （spectral:play、3 曲）、fb_ = ライブの変換（その場の分析つき）をヘッドホンで 3 曲、fc_ = 止まった話し手、fd_ = 2 ブロックずつ歩く話し手、fe_ = 3 ブロックずつ
#   （歩く速さ）。話し手の声はその場の分析つき（tools/exp/packs/live20 の _lr）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song vocal_name_of_love,rock_big,speech_ja_soseki suffix=_d pack=tools/exp/packs/run9
audio song vocal_name_of_love,speech_ja_soseki,speech_en_wordsworth suffix=_lr pack=tools/exp/packs/live20

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section fa
do execute as @a run tp @s 0 -60 0 0 0
audio start fa_1
audio mark fa_speech_ja_soseki
do execute as @a run spectral:play speech_ja_soseki_d
wait 23000
audio mark fa_vocal_name_of_love
do execute as @a run spectral:play vocal_name_of_love_d
wait 23000
audio mark fa_rock_big
do execute as @a run spectral:play rock_big_d
wait 23000
audio stop
audio analyze-all fa_speech_ja_soseki=song:speech_ja_soseki fa_vocal_name_of_love=song:vocal_name_of_love fa_rock_big=song:rock_big
audio export fa_speech_ja_soseki 160
audio export fa_vocal_name_of_love 160
audio export fa_rock_big 160
