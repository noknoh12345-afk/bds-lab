# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音そのものは `audio export` が Opus にして実行の成果物の
#   public/spectral/rec/ に公開する（notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。1 録音 = 1 区切り（失敗はその録音だけ）
# キュー 50（tools/exp/queue/q50_speech_joins.json、tools/exp/queue40.py）: 読み上げのつなぎ方（重ね 10・40 ms、母音を止める 0・20 ms）。素と素のつなぎ（既定: 重ね 25・止め 10）を変えて: 途切れ（プツッ）と潰れのどちらが聞き取りを下げるか

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_hello_o10,speech_cave_o10,speech_hello_o40,speech_cave_o40,speech_hello_c0,speech_cave_c0,speech_hello_c20,speech_cave_c20 pack=tools/exp/packs/q40

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear
do execute as @a run tp @s 0 -60 0 0 0

section join_o10_hello
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
audio start jno10_speech_hello
do execute as @a run spectral:play speech_hello_o10
wait 5997
do execute as @a run spectral:stop
audio stop
audio export jno10_speech_hello 160

section join_o10_cave
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
