# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 23 回目（tools/exp/make_app.py tools/exp/app23.json）: 歩く話し手の新しい既定（cut・1 tick に 2 音）を本物の歩きの速さで（3 ブロックずつ約 0.7 秒 =
#   4.3 ブロック/秒、毎歩が作り直し）。どれも 24 声・0.01。hg_ = 止まった話し手、ha_ / hf_ = cut 2 音、hb_ / he_ = cut 4 音、hc_ / hd_ = 1.2.0（hold 8 音）、
#   hh_ / hi_ = cut 2 音でライブの変換のまま / その場の分析つき（tools/exp/packs/live20 の夏目漱石）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9
audio song speech_ja_soseki suffix=_lb pack=tools/exp/packs/live20

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section hg
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true 0.01
do execute as @a run spectral:handover cut 1.5 2 0 2
do execute as @a run summon armor_stand talker 0 -60 3
audio start hg_1
audio mark hg_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d follow @s 24 "follow:talker"
wait 23000
audio stop
audio analyze-all hg_speech_ja_soseki=song:speech_ja_soseki
audio export hg_speech_ja_soseki 160
