# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音そのものは `audio export` が Opus にして実行の成果物の
#   public/spectral/rec/ に公開する（notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。1 録音 = 1 区切り（失敗はその録音だけ）
# キュー 28（tools/exp/queue/q28_velocity_curve.json、tools/exp/queue40.py）: 強さの曲線（8〜127 の 16 段、ピアノと弦）。強さ → 大きさが実機でどうなるか（三段の素の継ぎ目・16 bit の床に沈む弱さ）。大きさ 300 % と 200 %

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song velocity_ramp pack=tools/exp/packs/q40

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear
do execute as @a run tp @s 0 -60 0 0 0

section vel_phones
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
audio start vl_phones
do execute as @a run spectral:play velocity_ramp
wait 31100
do execute as @a run spectral:stop
audio stop
audio export vl_phones 160

section vel_speaker
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
audio start vl_speaker
do execute as @a run spectral:play velocity_ramp speaker
wait 31350
do execute as @a run spectral:stop
audio stop
audio export vl_speaker 160

section vel_l200
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 200 false 600
audio start vl_l200
do execute as @a run spectral:play velocity_ramp
wait 31100
do execute as @a run spectral:stop
audio stop
audio export vl_l200 160
