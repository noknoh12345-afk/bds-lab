# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 16 回目（tools/exp/make_app.py tools/exp/app16.json）: 新しい既定の確かめ — ライブの 3D を静かに作る（既定 on、聞こえる床は SILENT のまま）と
#   ふつうに作るを同じ run で。夏目漱石をライブの道で、ua_ / ub_ = 3 ブロック 静かに / ふつうに、uc_ / ud_ = 24 ブロック 静かに / ふつうに、
#   ug_ = 名札の防具立てを 3 ブロック前に置いて follow（止まったまま、静かに）、ue_ / uf_ = 防具立てを 0.5 秒ごとに 1 ブロック左右に 静かに / ふつうに

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section ua
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true
audio start ua_1
audio mark ua_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d near @s 24 "near:3,-58.5,0,overworld"
wait 23000
audio stop
audio analyze-all ua_speech_ja_soseki=song:speech_ja_soseki
audio export ua_speech_ja_soseki 160
