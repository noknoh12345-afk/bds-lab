# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 15 回目（tools/exp/make_app.py tools/exp/app15.json）: 3D の既定のレベル +10 dB（14 回目: 24 ブロックで ViSQOL 2.38 → 2.79）と、ライブの 3D の音を
#   静かに作る（/spectral:range 3 10 true: 定義の音量 0.1 の _3q の音を 10 倍に操る。13 回目: 高さ P_MIN で作っても、揃ったループの和がクリップして
#   約 0.1 秒の低い「トン」が残る）。夏目漱石をライブの道で、qd_ = 3 ブロック・ふつうに作る、qa_ / qb_ / qc_ = 3 / 24 / 46 ブロック・静かに作る
#   （頭の音・大きさ・届く距離が定義の音量で縮まないか）、qg_ = 24 ブロック・静かに・作る音量 16（qb_ が無音なら距離が縮んだ証拠）、
#   xg_ / xh_ = 46 ブロックで 0 / +10 dB（14 回目の lg_ / lh_ のくり返し: そこだけ +10 dB の ViSQOL が低かった）、
#   mg_ = 名札の防具立て「talker」を 3 ブロック前に置いて follow（止まったまま）、qf_ / qe_ = 同じ防具立てを 0.5 秒ごとに 1 ブロックずつ左右に
#   （約 1 秒ごとに音の作り直し = 引き継ぎ）、ふつうに / 静かに作って。14 回目は防具立てが無いときの kill がエラーで止まった（直した）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section qd
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 false
audio start qd_1
audio mark qd_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d near @s 24 "near:3,-58.5,0,overworld"
wait 23000
audio stop
audio analyze-all qd_speech_ja_soseki=song:speech_ja_soseki
audio export qd_speech_ja_soseki 160
