# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 18 回目（tools/exp/make_app.py tools/exp/app18.json）: 歩く話し手の作り直しの「トン」を小さく。話し手（名札の防具立て talker）の声を
#   定義の音量 0.1（_3q、1.2.0）・0.01（_3r）・0.001（_3s）の音で作り、その分だけ setVolume で上げる（spectral:range 3 10 true <定義の音量>）。
#   ta_ / tb_ = 止まったまま 0.1 / 0.001（setVolume を 1000 倍まで上げて同じ大きさで鳴るか）、wa_ wb_ wc_ wd_ we_ wf_ = 17 回目と同じ道を
#   0.1 / 0.01 / 0.001 / 0.001 / 0.01 / 0.1、wg_ / wi_ = 3 ブロック動いてから作り直す（spectral:handover hold 3）を 0.1 / 0.001

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section ta
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true 0.1
do execute as @a run spectral:handover hold 1.5
do execute as @a run summon armor_stand talker 0 -60 3
audio start ta_1
audio mark ta_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d follow @s 24 "follow:talker"
wait 23000
audio stop
audio analyze-all ta_speech_ja_soseki=song:speech_ja_soseki
audio export ta_speech_ja_soseki 160
