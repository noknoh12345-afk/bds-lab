# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。録音そのものは `audio export` が Opus にして実行の成果物の
#   public/spectral/rec/ に公開する（notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。1 録音 = 1 区切り（失敗はその録音だけ）
# キュー 61（tools/exp/queue/q61_noise_floor_warm.json、tools/exp/queue40.py）: 何も鳴らさない床と、温めそのものが聞こえるか。温めは音量 0.001 で 4 tick 鳴らす（p13 で 1 は聞こえると推測）。何も鳴らさない 20 秒（電話の床）、とても弱い音（強さ 4）、強さ 1 の音（温めだけが聞こえる）を温め 0.01 と 0.001 で

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song probe_quiet,probe_warmonly,probe_warmonly2 pack=tools/exp/packs/q40

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear
do execute as @a run tp @s 0 -60 0 0 0

section idle
audio start qi_idle
wait 20000
audio stop
audio export qi_idle 160

section quiet_w0
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 0
audio start qi_w0
do execute as @a run spectral:play probe_quiet
wait 14050
do execute as @a run spectral:stop
audio stop
audio export qi_w0 160

section quiet_again
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 0
audio start qi_again
do execute as @a run spectral:play probe_quiet
wait 14050
do execute as @a run spectral:stop
audio stop
audio export qi_again 160

section warm_only
do execute as @a run spectral:pitch 0
do execute as @a run spectral:speed 1
do execute as @a run spectral:volume 100
do execute as @a run spectral:sampler 2 250 28 300 false 600
do execute as @a run scriptevent spectral:warmcfg 0.01 4
audio start qi_warm01
do execute as @a run spectral:play probe_warmonly
wait 14350
do execute as @a run spectral:stop
audio stop
audio export qi_warm01 160
do execute as @a run scriptevent spectral:warmcfg 0.001 4
