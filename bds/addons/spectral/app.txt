# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 21 回目（tools/exp/make_app.py tools/exp/app21.json）: 作り直しのあと音節の頭が抜ける（19 回目: 引き継ぎの形によらず 300–350 ms）のは、
#   作り直しの間に 1 人の話し手が電話の 32 音を超えるからか。どれも 0.01・+10 dB。pa_ / pb_ = 止まった話し手 24 声 / 8 声、歩く話し手:
#   pc_ / ph_ = 8 声（8＋帯域 8 = 16 音: 作り直しの間も 32 音まで）、pd_ / pg_ = 24 声（1.2.0）、pe_ / pf_ = 24 声で古い音をすぐ止め 1 tick に 2 音ずつ
#   作り直す（spectral:handover cut 1.5 2 0 2: 32 音を超えない）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section pa
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true 0.01
do execute as @a run spectral:handover hold 1.5 2 0 8
do execute as @a run summon armor_stand talker 0 -60 3
audio start pa_1
audio mark pa_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d follow @s 24 "follow:talker"
wait 23000
audio stop
audio analyze-all pa_speech_ja_soseki=song:speech_ja_soseki
audio export pa_speech_ja_soseki 160
