# Spectral Player の測定リグ（本物のクライアントが鳴らした音を録る）: node lab.mjs app run -a spectral --device redroid
#   GitHub では Actions → redroid → Run workflow（addon: spectral, mode: run）。数はジョブのログの `[spectral rig]` の行、録音そのものは
#   `audio export` が Opus にして実行の成果物の public/spectral/rec/ に公開する（鍵なし: 管理パネルでサインインして聞ける・
#   notify.yml が Discord に送る・tools/rig/fetch_rec.py <run> で取る）。
# 17 回目（tools/exp/make_app.py tools/exp/app17.json）: 歩く話し手の引き継ぎ。名札の防具立て「talker」を 3 ブロック前で 2 ブロックずつ左右に
#   （1 歩 約 0.7 秒、毎歩が作り直し = 15・16 回目の約 2.2 秒ごとより 3 倍多い）。vt_ = 止まったまま、vh_ / wh_ = 古い音は最後の大きさのまま
#   （hold、1.2.0）、vs_ / ws_ = 古い音も声に付いていく（steer、spectral:handover steer）。どれも静かに作る・+10 dB（既定）

section title
audio info
stop
audio volume 0
launch
until title 600000
audio info
shot title

section prep
audio song speech_ja_soseki suffix=_d pack=tools/exp/packs/run9

section join
join
until joined 300000
wait 5000
do time set noon
do gamerule dodaylightcycle false
do weather clear

section vt
do execute as @a run tp @s 0 -60 0 0 0
do execute as @a run spectral:range 3 10 true
do execute as @a run spectral:handover hold
do execute as @a run summon armor_stand talker 0 -60 3
audio start vt_1
audio mark vt_speech_ja_soseki
do execute as @a run spectral:live_demo speech_ja_soseki_d follow @s 24 "follow:talker"
wait 23000
audio stop
audio analyze-all vt_speech_ja_soseki=song:speech_ja_soseki
audio export vt_speech_ja_soseki 160
