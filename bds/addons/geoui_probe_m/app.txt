# geoui_probe_m（bds/geoui/timeline/probe.mjs が作る）: GeoUI の Molang の長さ・深さの確かめ（pre_animation）。node bds/geoui/timeline/probe.mjs read <summary.json> が読む。
# 基準は何も出さない画面（world--probe-off）。テストを 1 つずつ出して写し、measure diff で変わった所（px。板のある所だけ）を数で出す。

section title
launch
until title 600000
shot title

section join
join
until joined 300000
wait 3000
do time set noon
do gamerule dodaylightcycle false
do gamerule sendcommandfeedback false
do gamerule commandblockoutput false
do weather clear
do tp @a 0 -60 0 0 90
do hud @a hide paperdoll
do hud @a hide touch_controls
do hud @a hide hotbar
do hud @a hide crosshair
do js for (const q of world.getAllPlayers()) q.setProperty('gprobm:v2', 0); 'probe off'
wait 6000
shot world--probe-off
absent clientlog \bERROR\b

section t01-ctl
do js for (const q of world.getAllPlayers()) q.setProperty('gprobm:v2', 1); 'probe 1'
wait 1500
shot t01-ctl
measure diff t01-ctl 0 170 1030 720
