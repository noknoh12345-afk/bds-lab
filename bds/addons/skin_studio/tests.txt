# Skin Studio 3 on a real BDS with real clients (op, spawn 0 -60 0, flat). Everything below is done by command only;
# the last sections use the forms once to show they still agree with the commands.

## every player gets a relay slot of their own
@A join
@B join
wait 1500
js [p('A').getProperty('skin_studio:slot') > 0, p('B').getProperty('skin_studio:slot') > 0, p('A').getProperty('skin_studio:slot') !== p('B').getProperty('skin_studio:slot'), p('A').getProperty('skin_studio:gen') > 0].join(',')
= true,true,true,true

## /studio:podium puts a podium on the block centre, facing the way asked
@A cmd /studio:podium 0 -60 6 north
~ 表彰台を置きました
wait 1000
js (() => { const [pd] = world.getDimension('overworld').getEntities({ type: 'skin_studio:podium' }); return [pd.location.x, pd.location.y, pd.location.z, Math.abs(Math.round(pd.getRotation().y))].join(' '); })()
= 0.5 -60 6.5 180

## /studio:rank puts people on the steps and /studio:rank without a player empties a step
@A cmd /studio:rank 1 A
~ 1位に A を入れました
@A cmd /studio:rank 2 B
~ 2位に B を入れました
wait 1500
js (() => { const [pd] = world.getDimension('overworld').getEntities({ type: 'skin_studio:podium' }); return [pd.getProperty('skin_studio:s1') === p('A').getProperty('skin_studio:slot'), pd.getProperty('skin_studio:g2') === p('B').getProperty('skin_studio:gen'), pd.nameTag.replace(/§./g, '').split('\n').join('|')].join(' '); })()
= true true 1位 A|2位 B
@A cmd /studio:rank 2
~ 2位を空けました
@A cmd /studio:rank 5 B
~ 順位は 1・2・3
wait 1500
js (() => { const [pd] = world.getDimension('overworld').getEntities({ type: 'skin_studio:podium' }); return [pd.getProperty('skin_studio:s2'), pd.nameTag.replace(/§./g, '')].join(' '); })()
= 0 1位 A

## a command block uses the podium next to it, execute as a podium uses that podium
gamemode creative A

= true true
setblock 3 -60 3 air

## an animal in front of the mirror gets a mirrored copy that follows it, and goes when it goes
summon cow 1.5 -60 9.5
effect @e[type=cow] slowness 30 255 true
wait 1500
js (() => { const d = world.getDimension('overworld'); const [c] = d.getEntities({ type: 'minecraft:cow', location: { x: 1.5, y: -60, z: 9.5 }, closest: 1 }); const r = d.getEntities({ type: 'skin_studio:m_cow' }).find((e) => Math.abs(e.getProperty('skin_studio:dx') - Math.round((c.location.x - 0.5) * 1000)) < 300); return [r.typeId, r.location.x, r.location.y, r.location.z, r.getProperty('skin_studio:slot') > 0].join(' '); })()
= skin_studio:m_cow 0.5 -60 12 true
kill @e[type=cow]
wait 2500
js world.getDimension('overworld').getEntities({ type: 'skin_studio:m_cow' }).length
= 0
