do effect @e[type=trader_llama] slowness 60 255 true
do effect @e[type=trader_llama] fire_resistance 60 1 true
do effect @e[type=trader_llama] water_breathing 60 1 true

section mobdc-trader_llama-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air

wait 2500

do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
