# 一括検証 3 回目（light-room-14-glow）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section room-14-glow
do fill -6 -60 -2 6 -55 -2 stone
do fill -6 -60 -2 -6 -55 12 stone
do fill 6 -60 -2 6 -55 12 stone
do fill -6 -55 -2 6 -55 11 stone
do summon pig 1.5 -60 7.5 150 0
do effect @e[type=pig] slowness 60 255 true
do setblock -4 -60 8 glowstone
do tp @a 2.5 -60 5.5 20 3
wait 4000
shot room-14-glow-mirror                        # room-14-glow: 鏡の中の明るさ・空・天気が正解と同じ
section room-14-glow-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air
do tp @a 2.5 -60 5.5 20 3
do camera @a set minecraft:free pos 2.5 -58.38 18.5 rot 3 160
wait 2500
shot room-14-glow-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して room-14-glow-mirror と比べる）
do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
section done
