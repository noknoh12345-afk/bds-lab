# 一括検証（anim）: 3.4.5 のまま、場面だけを変えた枝。結果は自分の鍵で封じる
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do time set noon
do weather clear
do gamemode creative @a
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section attack
do difficulty normal
do gamemode survival @a
do effect @a resistance 60 255 true
do effect @a regeneration 60 255 true
do tp @a 0.5 -60 7 0 0
do summon zombie -1.5 -60 5 0 0
do summon skeleton 2.5 -60 4 0 0
do summon pillager 3.5 -60 6 0 0
do effect @e[type=!player,family=monster] fire_resistance 60 1 true
record start
wait 8000
record stop                               # 殴る・弓を引く・クロスボウを構える: 鏡の中でも同じ腕の動き（左右反転）
do kill @e[family=monster]
do gamemode creative @a
do difficulty peaceful
section hurt-death
do summon pig -0.5 -60 8 150 0
do summon cow 1.5 -60 6.5 150 0
do tp @a 2.5 -60 5.5 20 3
wait 1500
record start
do damage @e[type=pig] 1
wait 1200
do damage @e[type=cow] 1
wait 1200
do damage @e[type=pig] 50
wait 1500
do damage @e[type=cow] 50
wait 2500
record stop                               # 赤くなる・倒れて消える: 鏡の中も同じ間合い
section ride
do summon horse 0.5 -60 8 0 0
do summon pig 2.5 -60 6.5 0 0
do tp @a 0.5 -60 8 0 0
do ride @a start_riding @e[type=horse,c=1] teleport_ride
do execute as @a at @s run studio:report
wait 1500
do camera @a set minecraft:free pos 3.5 -58 5 facing 0.5 -59 12
wait 1000
shot ride-mirror                          # 馬に乗った自分: 鏡の中でも馬の上（人の鏡像の乗り方）
record start
wait 4000
record stop
do camera @a clear
do ride @a stop_riding
do kill @e[type=horse]
do kill @e[type=pig]
section graze
do summon sheep -0.5 -60 8 150 0
do summon sheep 1.5 -60 7 210 0
do summon goat 2.5 -60 9 180 0
do tp @a 2.5 -60 5.5 20 3
record start
wait 10000
record stop                               # 草を食べる・首を振る・跳ねる（ヤギ）: 鏡の中で同じ
do kill @e[type=sheep]
do kill @e[type=goat]
section water
do fill -3 -61 4 3 -61 9 water
do summon cod -0.5 -61 6
do summon axolotl 1.5 -61 7
do summon squid 0.5 -61 5
do tp @a 2.5 -60 3 20 20
record start
wait 6000
record stop                               # 浅い水の中の魚・ウーパールーパー・イカ: 泳ぎ・ばたつき
do kill @e[type=cod]
do kill @e[type=axolotl]
do kill @e[type=squid]
do fill -3 -61 4 3 -61 9 grass_block
section done
