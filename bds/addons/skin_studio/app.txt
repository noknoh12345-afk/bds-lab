do effect @e[type=polar_bear] slowness 60 255 true
do effect @e[type=polar_bear] fire_resistance 60 1 true
do effect @e[type=polar_bear] water_breathing 60 1 true
do tp @a 1.5 -60 7.5 10 0
wait 3500
shot mobek-polar_bear-mirror                       # polar_bear: 鏡の中で左右反転・同じ模様・同じ大きさ・同じ動き
section mobek-polar_bear-truth
do tp @a 0.5 -60 10.5 0 0
do execute as @a at @s run studio:remove
do fill -4 -60 12 4 -56 12 air
do tp @a 1.5 -60 7.5 10 0
do camera @a set minecraft:free pos 1.5 -58.38 16.5 rot 0 170
wait 2500
shot mobek-polar_bear-truth                         # 正解: 鏡の面で折り返した位置から見た本物（左右反転して mobek-polar_bear-mirror と比べる）
do camera @a clear
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
wait 1500
