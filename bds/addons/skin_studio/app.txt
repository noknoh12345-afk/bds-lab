launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false

do time set noon
do weather clear
do gamemode creative @a

section setup
do fill -4 -60 12 4 -56 12 stone

wait 2500
