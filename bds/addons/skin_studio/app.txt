section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do gamerule doweathercycle false
do gamerule domobspawning false
do gamerule mobgriefing false
do gamerule doentitydrops false
do gamerule domobloot false
do time set noon
do weather clear
do gamemode creative @a
do effect @a resistance 100000 255 true
section setup
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do studio:mirror 0 -60 12 north
do replaceitem entity @a slot.weapon.mainhand 0 air
do tp @a 0.5 -60 8 0 0
wait 3000
shot setup                                # 鏡ができた
section clean579a
do kill @e[type=!player,family=!skin_studio]
wait 300
section clean579b
do kill @e[type=item]
section clean579c
do kill @e[type=xp_orb]
section attack-skeleton
do difficulty normal
do gamemode survival @a
do tp @a 0.5 -60 9 180 0
do summon skeleton 2.5 -60 4 0 0
do effect @e[type=!player,family=!skin_studio] fire_resistance 60 1 true
do camera @a set minecraft:free pos 4 -58.5 5 facing 0 -59.3 9.5
wait 500
record start
wait 9000
record stop                               # attack-skeleton: 本物と鏡の中の像が同じ動き（左右反転・同じ間合い）
shot attack-skeleton
do camera @a clear
