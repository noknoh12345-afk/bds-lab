# 本物のアプリで、画面の表彰台（GeoUI）と鏡を見る: node lab.mjs app run -a skin_studio（Actions の app: addon=skin_studio, mode=run）
section title
launch
until title 600000
shot title
section join
join
until joined 300000
wait 3000
do gamerule dodaylightcycle false
do time set noon
do weather clear
section podium
do studio:podium 0 -60 5 north
wait 1500
do execute as @a run studio:rank 1 @s
do tp @a 0.5 -60 0.5 facing 0.5 -58.5 5
wait 4000
shot podium                               # 1位に自分のスキン、2・3位は銀と銅の像（台の一部なので回転しない）
section screen
do tp @a 0.5 -60 0.5 0 90                 # 足元の地面だけを見る（雲も動かない）: 変わるのは画面の表彰台だけ
wait 3000
shot world
do execute as @a run studio:hud on
wait 4000
shot screen-podium                        # ← 右上の角に表彰台（GeoUI、高さの 30%、暗いカードの上）
changed 0.01
do execute as @a run studio:hudsize 60
wait 3000
shot screen-small                         # ← 同じ角のまま小さく
do execute as @a run studio:hudsize 100
perf 5000
do execute as @a run studio:hud off
wait 3000
shot screen-off
section mirror-field
do studio:mirror 20 -60 12 north           # 野原にそのまま立てた鏡
do setblock 21 -60 9 gold_block
do tp @a 20.5 -60 8.5 facing 20.5 -59 12
wait 4000
shot mirror-field                         # 枠の中にだけ自分が映る
do tp @a 26.5 -60 11.5 facing 20.5 -59 12
wait 3000
shot mirror-field-side                    # 横から: 枠の外に鏡像も鏡の中の金ブロックもはみ出さない
do tp @a 20.5 -60 15.5 facing 20.5 -59 12
wait 3000
shot mirror-field-back                    # 裏から: 何も映らない（鏡の中の金ブロックも見えない）
do execute as @a at @s run studio:remove
do setblock 21 -60 9 air
section mirror-pair
do studio:mirror 24 -60 24 north
do studio:mirror 27 -60 24 north
do tp @a 21.5 -60 19.5 facing 27.5 -59 24
wait 4000
shot mirror-pair                          # 並べた 2 枚の鏡を斜めから: どちらの窓にも、その鏡の鏡像だけ（隣の窓に入り込まない）
do tp @a 30.5 -60 19.5 facing 24.5 -59 24
wait 3000
shot mirror-pair-other                    # 反対の斜めからも同じ
do execute as @a at @s run studio:remove
do execute as @a at @s run studio:remove
section mirror
do fill -4 -60 12 4 -56 12 stone
do fill -1 -60 12 1 -58 12 air
do setblock 2 -60 9 gold_block
do setblock -2 -60 10 oak_planks
do setblock -2 -59 10 oak_log ["pillar_axis"="x"]
do setblock 2 -59 9 crafting_table
do studio:mirror 0 -60 12 north
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
wait 5000
shot mirror                               # 自分が左右反転で映り、金ブロック・板材・横倒しの原木（切り口が横）・作業台（天板が上）が窓の奥に描かれる（本物は置かない）
do tp @a 0.5 -60 11.3 facing 0.5 -58.4 12
wait 3000
shot mirror-close                         # ガラスの目の前（0.7）: 自分も金ブロックも板材も消えずに映る
do tp @a 0.5 -60 11.85 facing 0.5 -58.4 12
wait 2000
shot mirror-pushed                        # ガラスに入り込んでも前へ戻されて、映ったまま
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
wait 2000
record start
walk left 1
walk right 1
look left 300 40
look right 300 40
record stop                               # 歩く・振り向く鏡像がカクつかないか（recordings/rec-1.mp4）
shot mirror-after
do camera @a set minecraft:third_person
wait 3000
shot mirror-3p                            # 三人称（後ろから）: 自分の背中の先の窓に、こちらを向いた自分が左右反転で映る
look left 300 40
wait 1500
shot mirror-3p-turned                     # 三人称で振り向いても鏡像の向きが合う
do camera @a clear
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
wait 2000
shot mirror-1p-again                      # 一人称に戻しても映る
do tp @a 5.5 -60 8.5 facing 0.5 -59 12
wait 3000
shot mirror-blocks                        # 自分は窓の外: 鏡の中のブロック（金・板材・横倒しの原木・作業台）だけを斜めから
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
wait 1000
do summon boat 0.5 -60 9.5
do ride @a start_riding @e[type=boat,c=1] teleport_rider
wait 3000
shot mirror-ride                          # ボートに座った自分が、低く座った姿で映る（立ったままではない）
do ride @a stop_riding
do kill @e[type=boat]
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
wait 2000
section mirror-mobs
do summon cow 1.5 -60 10.5
do summon sheep -1.5 -60 9.5
do summon villager 0.5 -60 8.5
do summon pig -0.5 -60 10.5
do summon chicken 1.5 -60 8.5
do summon zombie -1.5 -60 8.5
do effect @e[type=!player] slowness 60 255 true
do effect @e[type=zombie] fire_resistance 60 1 true
do tp @a 3.5 -60 6.5 facing 0.5 -59 12
wait 5000
shot mirror-mobs                          # 牛・羊・村人・豚・鶏・ゾンビも左右反転で窓の奥に映る（色・模様・首の向き・歩きも本物と同じ）
do camera @a set minecraft:third_person
wait 3000
shot mirror-mobs-3p                       # 三人称でも同じ
do camera @a clear
do tp @a 0.5 -60 11 facing 0.5 -59 12
do tp @e[type=cow] 0.5 -60 6.5
wait 3000
shot mirror-mob-behind                    # 背中側で画面に入っていない牛も映る（サーバー経由の予備の動き）
do kill @e[type=!player]
section mirror-debug
do summon cow 1.5 -60 10.5
do effect @e[type=cow] slowness 60 255 true
do tp @a 0.5 -60 7.5 facing 0.5 -59 12
do execute as @a run studio:debug on
wait 4000
shot mirror-debug                         # 自分の鏡像と牛の鏡像の上に緑の点（中継で映っている）。黄の三角・紫の四角・赤の × は無い
do execute as @a run studio:debug off
do kill @e[type=cow]
section mirror-paused
do setblock 0 -59 12 glass
wait 3000
shot mirror-paused                        # 窓をふさぐと鏡は止まる: 自分も鏡の中のブロックも映らない
do setblock 0 -59 12 air
wait 4000
shot mirror-resumed                       # 空けると元どおり映る
section done
absent clientlog \bERROR\b                # コンテンツログに、パック・JSON UI・スクリプトのエラーが無い
