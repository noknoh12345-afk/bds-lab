## bench: every tower level shoots a dummy at the wiki's Damage ÷ Firerate (±5%), hits inside its range and never outside

!~ "ev":"bench_bad"

## detection: a tower without hidden/lead/flying detection lets those enemies pass; with it, they die
tds:stop
@A cmd /tds:play Easy
tds:hold true
tds:speed 10
@A cmd /tds:place Scout 0 -60 -13
@A cmd /tds:cash 5000
tds:spawn Hidden
until "ev":"leak","id":\d+,"name":"Hidden" 30000
!~ "ev":"enemy_kill","id":\d+,"name":"Hidden"
@A cmd /tds:upgrade 1
@A cmd /tds:upgrade 1
~ @A Scout → Lv2
tds:spawn Hidden
until "ev":"enemy_kill","id":\d+,"name":"Hidden","by":1 30000
tds:spawn Lead
until "ev":"leak","id":\d+,"name":"Lead" 30000
!~ "ev":"enemy_kill","id":\d+,"name":"Lead"
tds:spawn Balloon
until "ev":"leak","id":\d+,"name":"Balloon" 30000
!~ "ev":"enemy_kill","id":\d+,"name":"Balloon"
tds:speed 1

## leaks: an enemy reaching the base removes its remaining health from base HP; HP 0 loses
tds:stop
@A cmd /tds:play Easy
tds:speed 20
until "ev":"leak","id":\d+,"name":"Normal","dmg":4,"baseHP":196 60000
until "ev":"game_end","result":"lose" 120000
until "ev":"reward","player":"A","result":"lose" 5000
tds:speed 1

## lobby: profile, shop with coins, loadout, quests, elevator to a match
tds:stop
@A cmd /tds:profile
~ @A Lv\d+ exp \d+ coins \d+ gems \d+ owned \d+ loadout
@A cmd /tds:grant 1000 0 0
@A cmd /tds:buy Scout
~ @A Already own Scout
@A cmd /tds:buy War_Machine
~ @A War Machine is not sold \(Exclusive\)
@A cmd /tds:buy Freezer
~ @A Unlocked Freezer|@A Already own Freezer
@A cmd /tds:equip 2 Freezer
~ @A Loadout: Scout, Freezer
@A cmd /tds:quests
~ @A Defeat 300 enemies \d+/300 .*Win a match 1/1
@A cmd /tds:grant 0 0 0
@A cmd /tds:queue Hardcore
~ @A Hardcore needs level 50
@A cmd /tds:queue Casual
~ @A Elevator Casual: 1/4, departs in 20s
tds:depart Casual
until "ev":"match_start","mode":"Casual" 5000
~ "ev":"elevator_depart","mode":"Casual","players":\["A"\]

## restart: the match, its towers and wave survive a server restart
tds:hold true
@A cmd /tds:place Freezer 0 -60 -14
~ @A Placed #\d+ Freezer
restart
@A join
tds:state
~ "ev":"restored","mode":"Casual","wave":0,"enemies":0,"towers":1
~ "towers":\["\d+:Freezer:0"\]
