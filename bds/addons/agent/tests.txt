## tp moves me to x y z
@A join
@A cmd /lab:tp 5 -60 5
~ @A tp 5 -?\d+ 5
js [Math.floor(p('A').location.x), Math.floor(p('A').location.z)].join(' ')
= 5 5

## move is relative
@A cmd /lab:move 2 0 3
~ @A move 7 -?\d+ 8
js [Math.floor(p('A').location.x), Math.floor(p('A').location.z)].join(' ')
= 7 8

## look issues a view change (a real client's view is client-authoritative, so the value is the client's)
@A cmd /lab:look 90 10
~ @A look -?\d+ -?\d+

## place then break a block
@A cmd /lab:place 3 -60 3 stone
~ @A place stone
js world.getDimension('overworld').getBlock({x:3,y:-60,z:3}).typeId
= minecraft:stone
@A cmd /lab:break 3 -60 3
~ @A break air
js world.getDimension('overworld').getBlock({x:3,y:-60,z:3}).typeId
= minecraft:air

## say broadcasts to chat
@A cmd /lab:say hello
~ A: hello

## query reports position, rotation and player count
@A cmd /lab:query
~ @A self 7 -?\d+ 8 yaw -?\d+ pitch -?\d+ players 1

## face points the view at a target
@A cmd /lab:face 0 -60 0
~ @A face -?\d+ -?\d+

## goto walks me to a target with no touch (continuous server teleport)
@A cmd /lab:goto 12 9
~ @A goto 12 9
until arrived 15000
js [Math.round(p('A').location.x), Math.round(p('A').location.z)].join(' ')
= 12 9

## stop halts walking
@A cmd /lab:goto -20 -18
@A cmd /lab:stop
~ @A stop
